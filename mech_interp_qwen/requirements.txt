# Deps for the Qwen2.5-VL CD mechanistic battery.
#
# torch/torchvision are deliberately NOT pinned: Lightning Studio images ship a CUDA-matched
# torch, and reinstalling it from PyPI is the single most common way to end up with a CPU-only
# or ABI-mismatched build. run_all_lightning.sh creates the venv with --system-site-packages so
# the Studio's torch is reused, and errors out if it is missing or older than 2.1.
#
# transformers is pinned: Qwen2.5-VL landed in 4.49 and its internals (module layout, mask
# construction, layer return types) have churned since. 4.51.3 is the version this port was
# written and reasoned against. The code still resolves submodules by search rather than by
# hardcoded path, so a bump is survivable -- but run selfcheck.py after any bump.
transformers==4.51.3
tokenizers>=0.21,<0.22
accelerate>=0.34
safetensors>=0.4.3
huggingface_hub>=0.26
hf_transfer>=0.1.6

pillow>=10.0
