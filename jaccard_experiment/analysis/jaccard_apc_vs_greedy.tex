\begin{table}[ht]
\centering
\caption{BPE-token Jaccard similarity of APC outputs against Greedy Search and Direct Sampling
at four representative $\beta$ values, evaluated on LLaVA-v1.5-7B across 60 LLaVA-Bench
(In-the-Wild) questions. In every case, APC outputs are more similar to Greedy Search than to
Direct Sampling, showing that even at low $\beta$, APC pulls the output distribution towards
greedy rather than preserving the diversity of direct sampling.}
\label{tab:jaccard_apc_greedy_vs_sample}

\setlength{\tabcolsep}{12pt}
\renewcommand{\arraystretch}{1.2}

\begin{tabular}{ccc}
\toprule
\textbf{APC $\beta$}
  & \textbf{Jaccard vs.\ Greedy}
  & \textbf{Jaccard vs.\ Direct Sampling} \\
\midrule
0.075 & \textbf{0.3689} & 0.3069 \\
0.100 & \textbf{0.4027} & 0.3135 \\
0.200 & \textbf{0.4032} & 0.3117 \\
0.300 & \textbf{0.4182} & 0.3068 \\
\bottomrule
\end{tabular}

\medskip
\noindent
\textbf{Claim:} When direct sampling is combined with the Adaptive Plausibility Constraint
(APC), the outputs become more similar to greedy search than to direct sampling itself,
even at small values of $\beta$. This shows that APC does not preserve the diversity of
direct sampling --- instead, it systematically shifts the output distribution towards the
deterministic greedy solution.

\end{table}

\begin{table}[ht]
\centering
\caption{BPE-token Jaccard similarity between VCD (Visual Contrastive Decoding) outputs and
three reference conditions --- Greedy Search, APC at its default $\beta{=}0.1$, and Direct
Sampling --- evaluated on LLaVA-v1.5-7B across 60 LLaVA-Bench (In-the-Wild) questions.}
\label{tab:jaccard_vcd}

\setlength{\tabcolsep}{14pt}
\renewcommand{\arraystretch}{1.2}

\begin{tabular}{lc}
\toprule
\textbf{Comparison} & \textbf{Mean Jaccard (BPE)} \\
\midrule
VCD vs.\ Greedy Search      & \textbf{0.3977} \\
VCD vs.\ APC ($\beta=0.1$)  & 0.3705 \\
VCD vs.\ Direct Sampling    & 0.3055 \\
\bottomrule
\end{tabular}

\medskip
\noindent
\textbf{Explanation:} VCD outputs are most similar to Greedy Search (Jaccard $= 0.3977$) and
least similar to Direct Sampling (Jaccard $= 0.3055$), even though VCD is built on top of
direct sampling. This happens because VCD internally applies APC at $\beta{=}0.1$ to filter
the output distribution at every decoding step. As shown in the table above, APC shifts direct
sampling towards greedy search even at small $\beta$ values. Since VCD inherits this APC step,
its outputs end up closer to greedy search than to the direct sampling distribution it was
intended to improve upon.

\end{table}
